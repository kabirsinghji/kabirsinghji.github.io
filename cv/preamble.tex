\documentclass[10pt,letterpaper]{article}

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage{charter}
\usepackage[margin=0.7in,top=0.6in,bottom=0.75in]{geometry}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{tabularx}
\usepackage{fancyhdr}
\usepackage{lastpage}
\usepackage{microtype}
\usepackage{needspace}
\usepackage[hidelinks]{hyperref}

\setlength{\parindent}{0pt}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[R]{\footnotesize\thepage\ of \pageref{LastPage}}
\widowpenalty=10000
\clubpenalty=10000

%---- Section headings: bold small caps + full-width rule -------
\titleformat{\section}
  {\fontsize{13}{15}\selectfont\bfseries\scshape}{}{0em}{}[\vspace{-6pt}\rule{\textwidth}{0.8pt}]
\titlespacing*{\section}{0pt}{9pt}{3pt}

%---- \entry{left block}{dates} --------------------------------
% Left block wraps freely; dates stay pinned top-right.
% Tune \entrygap to control visual density (try 2pt–6pt).
\newlength{\entrygap}\setlength{\entrygap}{5pt}
\newcommand{\entry}[2]{%
  \par\vspace{\entrygap}%
  \noindent
  \begin{minipage}[t]{0.795\textwidth}\raggedright #1\end{minipage}%
  \hfill
  \begin{minipage}[t]{0.19\textwidth}\raggedleft\small #2\end{minipage}%
  \par\penalty500
}

%---- Subheading inside a section ------------------------------
\newcommand{\subhead}[1]{%
  \par\vspace{10pt}\noindent$\diamond$\ {\fontsize{11}{13}\selectfont\itshape #1}\ $\diamond$\par\vspace{3pt}}

%---- Bullet list used under an \entry -------------------------
\newlist{cvitems}{itemize}{1}
\setlist[cvitems]{label=\textendash,leftmargin=1.3em,topsep=1pt,
                  itemsep=0pt,parsep=0pt,before=\small}

%---- Presentations: citation, then venue / format / year rows --
% \pres[lines]{citation} -- optional arg reserves space so the citation and its
% venue table are never split across a page break (tabularx cannot break).
\newcommand{\pres}[2][6]{\needspace{#1\baselineskip}%
  \par\vspace{6pt}\noindent\raggedright #2\par\vspace{1pt}}
\newcommand{\venues}[1]{%
  \begingroup
  \noindent\hspace*{1.3em}\small
  \begin{tabularx}{\dimexpr\textwidth-1.3em\relax}{@{}Xlr@{}}#1\end{tabularx}\par
  \endgroup}

%---- Flat lists (certifications) ------------------------------
\newlist{flatlist}{itemize}{1}
\setlist[flatlist]{label=\textbullet,leftmargin=1.3em,topsep=2pt,
                   itemsep=0pt,parsep=0pt}
