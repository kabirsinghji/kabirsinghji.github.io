%==============================================================
%  Kabir Singh — CV layout and styles (EB Garamond, compiled with XeLaTeX).
%  Every cv*.tex (generated from data/cv.json, one per profile)
%  starts with \input{preamble}. Change fonts, spacing and the
%  entry macros here; the .docx copies these choices by hand.
%==============================================================
\documentclass[12pt,a4paper]{article}

% XeLaTeX (the GitHub Action runs latexmk -xelatex). Georg Duffner's EB Garamond (v0.016), loaded from
% ./fonts: his 08 optical size below 10.1pt, the 12 size above, as he
% recommends; true small caps are built in. Duffner never finished a bold
% (his beta lacks curly quotes), so bold is the SemiBold from Octavio Pardo's
% completion of the same design, set with Duffner's old-style figures.
% Set at 12pt, which reads about the size of Charter at 11pt; the date
% column uses full-height (lining) figures so dates scan easily.
\usepackage{fontspec}
\setmainfont{EBGaramond12-Regular}[
  Path=fonts/, Extension=.otf, Ligatures=TeX,
  ItalicFont=EBGaramond12-Italic, BoldFont=EBGaramond-SemiBold,
  BoldFeatures={Numbers=OldStyle},
  UprightFeatures={SizeFeatures={{Size=-10.1,Font=EBGaramond08-Regular},{Size=10.1-}}},
  ItalicFeatures={SizeFeatures={{Size=-10.1,Font=EBGaramond08-Italic},{Size=10.1-}}}]
\newcommand{\datefigs}{\addfontfeatures{Numbers=Lining}}
\usepackage[margin=1.8cm,includefoot,footskip=0.8cm]{geometry}
\usepackage{xcolor,enumitem,array,tabularx,needspace,fancyhdr,lastpage,microtype,multicol,graphicx}
\usepackage[hidelinks]{hyperref}

\definecolor{promptgray}{gray}{0.46}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\setlist[itemize]{leftmargin=1.3em,topsep=1pt,itemsep=0pt,parsep=0pt,partopsep=0pt}
\setlist{beginpenalty=10000,before=\global\cvkeepfalse}
\setlength{\tabcolsep}{0pt}
\emergencystretch=3em % room to loosen a line rather than let tied last words run into the margin
% no hanging lines: never one line of a paragraph alone at the top or bottom of
% a page, and no paragraph ending on a lone word or half a word. build.py ties
% each paragraph's last words and marks them: from \cvkeep a word may hyphenate
% only if 5+ letters go to the next line; from \cvlast nothing hyphenates;
% \cvkeepend restores the defaults
\widowpenalty=10000 \clubpenalty=10000
\newlanguage\cvnohyphenation % a language with no patterns: nothing hyphenates
\newcommand{\cvkeep}{\leavevmode\xdef\cvlanguage{\the\language}\xdef\cvrhm{\the\righthyphenmin}%
  \righthyphenmin=5 \setlanguage\language}
\newcommand{\cvlast}{\language=\cvnohyphenation\relax}
\newcommand{\cvkeepend}{\language=\cvlanguage\relax\righthyphenmin=\cvrhm\relax}
% a heading or entry title is glued to what follows it: the next entry skips its
% own break point, so no heading or title is left alone at the foot of a page
\newif\ifcvkeep
\newcommand{\cvbreak}[1]{\par\ifcvkeep\nopagebreak\else\needspace{#1}\fi\global\cvkeepfalse}
\newcommand{\cvglue}{\par\ifcvkeep\nopagebreak\fi\global\cvkeepfalse}
\pagestyle{fancy}\fancyhf{}
\renewcommand{\headrulewidth}{0pt}\renewcommand{\footrulewidth}{0pt}
\setlength{\headheight}{14pt}
\setlength{\multicolsep}{2pt}
\fancyfoot[R]{\footnotesize Page \thepage\ of \pageref{LastPage}}
% the left footer ("Kabir Singh · Curriculum Vitae · <as of>") is set in each cv*.tex

%---- Section headings: true small caps + rule -------------------
% EB Garamond has real small caps, so the faux version isn't needed.
\newcommand{\fauxsc}[1]{\textsc{#1}}
% each heading also adds a PDF bookmark (the outline in a viewer's sidebar); subheadings nest under it
\hypersetup{bookmarksopen=true}
\newcounter{cvmark}
\newcommand{\cvsection}[1]{\global\cvkeepfalse\needspace{3\baselineskip}\vspace{9pt}\stepcounter{cvmark}\pdfbookmark[0]{#1}{cvmark.\thecvmark}{\fauxsc{#1}}\par\nopagebreak\vspace{2pt}\hrule height .45pt\nopagebreak\vspace{5pt}\global\cvkeeptrue}
\newcommand{\subhead}[1]{\cvbreak{2\baselineskip}\vspace{4pt}\stepcounter{cvmark}\pdfbookmark[1]{#1}{cvmark.\thecvmark}{\itshape #1}\par\nopagebreak\vspace{2pt}\global\cvkeeptrue}

%---- Entries ----------------------------------------------------
% One gap before every entry; tune it to change the density.
\newlength{\entrygap}\setlength{\entrygap}{4pt}
% \headentry{title}{dates}{second line}{right of second line}
%   two lines: bold title + dates, then organisation + (second date line, location)
\newcommand{\headentry}[4]{\cvbreak{2\baselineskip}\vspace{\entrygap}\noindent\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X@{\hspace{1em}}>{\datefigs}r@{}}\textbf{#1}&#2\\[-1pt]#3&#4\end{tabularx}\par\global\cvkeeptrue}
% \lineentry{text}{dates}: one line, as in the two-page CV
\newcommand{\lineentry}[2]{\cvbreak{1.3\baselineskip}\vspace{\entrygap}\noindent\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X@{\hspace{1em}}>{\datefigs}r@{}}#1 & #2\end{tabularx}\par}
% \roleline{text}{dates}: one of several roles under an entry (no gap above)
\newcommand{\roleline}[2]{\par\nopagebreak\noindent\begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X@{\hspace{1em}}>{\datefigs}r@{}}#1 & #2\end{tabularx}\par}
% \project{title}: an italic project title under a research position
\newcommand{\project}[1]{\cvbreak{2\baselineskip}\vspace{1pt}\noindent{\itshape #1}\par}
% grey note to self, printed only by profiles with prompts on
\newcommand{\prompt}[1]{\begin{itemize}[leftmargin=1.45em]\item[\textcolor{promptgray}{→}]\textcolor{promptgray}{\itshape #1}\end{itemize}}
% label / text rows (coursework, skills); 3.8cm fits the longest label,
% "Computational biology", and matches the .docx label tab; the text is
% ragged-right so the narrow column never spaces out
\newcommand{\course}[2]{\cvglue\noindent\begin{tabularx}{\linewidth}{@{}>{\itshape}p{3.8cm}@{\hspace{0.5em}}>{\raggedright\arraybackslash}X@{}}#1 & #2\end{tabularx}\par}
% presentations: citation, then one row per venue
\newcommand{\pres}[1]{\cvbreak{4\baselineskip}\vspace{\entrygap}\noindent #1\par\global\cvkeeptrue}
\newcommand{\venue}[3]{\cvglue\noindent\hspace*{1.0em}\begin{tabularx}{\dimexpr\linewidth-1em\relax}{@{}X@{\hspace{1em}}>{\datefigs}r@{}}#1 (\emph{#2}) & #3\end{tabularx}\par}
